\section{مجموعه}
مجموعه یکی از مفاهیم ریاضی است که تعریف ندارد. منظور از یک مجموعه دسته‌ای از اشیای کاملاً مشخص است که هر شیء مفروض نسبت به این مجموعه یکی از دو وضع متمایز زیر را داراست:
\begin{enumerate}
	\item 
	آن شیء متعلق به مجموعه باشد.
	\item 
	آن شیء متعلق به مجموعه نباشد.
\end{enumerate}
اشیایی که مجموعه را تشکیل می‌دهند به نام عنصرهای (عضوهای) مجموعه خوانده می‌شوند. مجموعه‌ها به طریق نام بردن عضوها یا با بیان خاصیتی معین مشخص می‌شوند. هیچ نوع محدودیتی در نوع عضوهای یک مجموعه وجود ندارد، اما هر گروهی از اشیا که مشخص نباشد، مجموعه نیست.
\begin{example}
	موارد زیر مجموعه هستند:
	\begin{enumerate}
		\item 
		مجموعه‌ی عددهای ۱، ۴، ۹ و ۱۰ (نام بردن عنصرهای متعلق به مجموعه)
		\item 
		مجموعه‌ی عددهای ۱، ۲، ۳ و ... (عددها به همین ترتیب ادامه دارند. به دلیل زیاد بودن عضوها، مجموعه به این صورت نوشته می‌شود.)
		\item 
		مجموعه‌ی روزهای هفته (بیان خاصیتی معین)
	\end{enumerate}
	اما موارد زیر مجموعه نیستند:
	\begin{enumerate}
		\item 
		سه نفر از شاعران ایرانی (بیان خاصیتی نامعین)
		\item 
		دسته‌ای که ویژگی عنصرهایش این است که هر عنصر متعلق به خودش نیست. (تناقض دارد.)
		\item 
		دسته‌ای که هر عضوش حداقل یک عضو دارد.
	\end{enumerate}
\end{example}
\smallskip
مجموعه‌ی $S$ را در نظر بگیرید. اگر عنصر $x$ متعلق به مجموعه‌ی $S$ باشد می‌نویسیم
$x \in S$
و اگر متعلق نباشد می‌نویسیم
$\neg(x \in S)$ یا برای سادگی $x \notin S$.
برای نوشتن عنصرهای مجموعه یا بیان ویژگی معین، آن را داخل $\{\}$ می‌گذارند. مجموعه‌ی $x$هایی که $x$ دارای ویژگی مشخص $p(x)$ است به صورت زیر نمایش داده می‌شود.
$$S=\{x \ | \ x \in A \ \wedge \ p(x)\} = \{x \in A \ | \ p(x)\}$$
در عبارت بالا، $‌‌A$ مجموعه است. مهم است که $x$ها از یک مجموعه انتخاب شوند، وگرنه ممکن است $S$ مجموعه نباشد.

\begin{example}
	مجموعه‌ی
	$A = \{m \ | \ m \in \mathbb{Z} \ \wedge \ 10<m<14\}$
	را با عنصرهایش به صورت
	$A = \{11, 12, 13\}$
	نمایش می‌دهیم. برای مثال داریم:
	$11 \in A$ و
	$14 \notin A$.
\end{example}
\smallskip
مجموعه‌ای را که تعداد اعضای آن متناهی باشد، مجموعه‌ی متناهی می‌نامیم و مجموعه‌ای که متناهی نباشد، نامتناهی می‌نامیم. تعداد عضوهای یک مجموعه‌ی متناهی را عدد اصلی (عدد کاردینال) آن مجموعه می‌نامیم. عدد کاردینال مجموعه‌ی $S$ را با
$|S|$ یا $n(S)$
نشان می‌دهیم.
\begin{example}
	عدد اصلی مجموعه‌ی
	$D = \{x \ | \ x \in \mathbb{Z} \ \wedge \ x^2<16\}$
	را بیابید.
	\\ \textbf{پاسخ:}
	$$D = \{-3,-2,-1,0,1,2,3\}$$
	بنابراین عدد کاردینال مجموعه‌ی $D$ برابر با ۷ است.
\end{example}
\smallskip
مجموعه‌ای وجود دارد که هیچ عنصری متعلق به آن نیست؛ به بیان ریاضی:
$$\exists x \ \forall y \ \neg(y \in x)$$
چنین مجموعه‌ای یکتاست و آن را
\textbf{تهی}
می‌نامیم و با نماد
$\emptyset$ یا $\varnothing$
نشان می‌دهیم که معادل
$\{\}$
است. این مجموعه را نباید با
$\{\varnothing\}$
اشتباه گرفت. \\
\textbf{تساوی دو مجموعه:}
دو مجموعه‌ی $A$ و $B$ برابرند اگر و تنها اگر عنصرهایشان یکی باشد و می‌نویسیم
$A=B$.
اگر $A$ و $B$ برابر نباشند، می‌نویسیم
$A \neq B$.
با توجه به این تعریف، نتیجه می‌شود که تغییر ترتیب عنصرهای یک مجموعه، آن مجموعه را تغییر نمی‌دهد. همچنین، تکرار عنصرها در یک مجموعه، آن مجموعه را تغییر نمی‌دهد.
\begin{example}
	مجموعه‌های زیر را در نظر بگیرید.
	$$A=\{3,7,9\}, \ B=\{9,3,7\}, \ C=\{2,5,2,8\}, \ D=\{2,5,8\}$$
	داریم:
	$A=B$ و $C=D$.
\end{example}
\begin{example}
	اگر دو مجموعه‌ی
	$\{3,\{z-1,y+1\},\{z\}\}$
	 و
	$\{\{2,1\},\{y+1\},x+1\}$
	 برابر باشند، آنگاه مقدارهای
	$x$، $y$ و $z$
	را به دست آورید.
	\\ \textbf{پاسخ:}
	\[
	\begin{cases}
		x+1=3 \\
		y+1=z \\
		\{2,1\}=\{z-1,y+1\}
	\end{cases}
	\Rightarrow
	x=2 \ , \ y=1 \ , \ z=2
	\]
\end{example}
\smallskip
هرگاه بدانیم که $A$ و $B$ مجموعه هستند، برای این‌که نشان دهیم که
$A=B$،
کافی است نشان دهیم که برای هر عنصر دلخواه
$x_0 \in A$
داریم
$x_0 \in B$
و برای هر عنصر دلخواه
$x_0 \in B$
داریم
$x_0 \in A$.
\\
\textbf{زیرمجموعه:}
مجموعه‌ی $A$ را زیرمجموعه‌ی $B$ می‌گوییم هرگاه هر عنصر متعلق به $A$، متعلق به $B$ باشد و به صورت
$A \subseteq B$
نشان می‌دهیم؛ به بیان ریاضی:
$$A \subseteq B  \iff \forall x \ (x \in A \rightarrow x \in B)$$
اگر $A$ زیرمجموعه‌ی $B$ نباشد، آن را به صورت
$A \nsubseteq B$
نمایش می‌دهیم. \\
تساوی دو مجموعه را می‌توان به این صورت نیز بیان کرد که
$A=B$
اگر و تنها اگر
$A \subseteq B$ و $B \subseteq A$.
\begin{example}
	فرض کنید
	$A=\{1,2,3,4,5\}$ و $B=\{2,3,4\}$.
	برای مثال داریم:
	$$B \subseteq A \ , \quad \{1\} \subseteq A \ , \quad A \nsubseteq B \ , \quad \{2\} \notin B$$
\end{example}
\begin{theorem}
	مجموعه‌ی تهی زیرمجموعه‌ی همه‌ی مجموعه‌هاست؛ به بیان ریاضی:
	$$\forall x \quad \varnothing \subseteq x$$
	\dfth \\ \textbf{برهان:}
	باید نشان دهیم که
	$\forall y \ \forall x \ (y \in \varnothing \rightarrow y \in x)$.
	برای این منظور مجموعه‌های دلخواه
	$x_0$ و $y_0$
	را در نظر می‌گیریم. باید نشان دهیم عبارت
	$y_0 \in \varnothing \rightarrow y_0 \in x_0$
	درست است. گزاره‌ی
	$y_0 \in \varnothing$
	نادرست است، پس ارزش کل گزاره به انتقای مقدم درست است.
	\qed
\end{theorem}
\begin{theorem}
	هر مجموعه‌ای زیرمجموعه‌ی خودش است. یعنی برای هر مجموعه‌ی $A$ داریم
	$A \subseteq A$.
	\\ \dfth \\ \textbf{برهان:}
	باید نشان دهیم که
	$\forall x \ (x \in A \rightarrow x \in A)$.
	واضح است که این جمله درست است.
	\qed
\end{theorem}
\begin{theorem}
	برای سه مجموعه‌ی $A$، $B$ و $C$ اگر
	$A \subseteq B$ و $B \subseteq C$،
	آنگاه
	$A \subseteq C$.
	\\ \dfth \\ \textbf{برهان:}
	باید نشان دهیم که
	$\forall x \ (x \in A \rightarrow x \in C)$.
	عنصر دلخواه
	$x_0 \in A$
	را در نظر بگیرید. طبق فرض
	$A \subseteq B$،
	داریم
	$x_0 \in B$.
	حال طبق فرض
	$B \subseteq C$
	داریم
	$x_0 \in C$.
	بنابراین، هر عنصر متعلق به $A$ متعلق به $C$ است. پس
	$A \subseteq C$.
	\qed
\end{theorem}
\smallskip
اگر
$A \subseteq A$
ولی
$A \neq B$،
آنگاه $A$ زیرمجموعه‌ی سره (محض) $B$ نامیده می‌شود و آن را به صورت
$A \subset B$
نشان می‌دهیم.
\begin{example}
	یک مجموعه‌ی سه عضوی بنویسید که از هر دو عضو آن یکی زیرمجموعه‌ی دیگری باشد.
	\\ \textbf{پاسخ:}
	$$A = \{\varnothing, \{1\}, \{1,2\}\}$$
\end{example}
\begin{example}
	فرض کنید
	$A = \{n \ | \ n=5k+2 \ \wedge \ k \in \mathbb{Z}\}$ و
	$B = \{n \ | \ n=5k-3 \ \wedge \ k \in \mathbb{Z}\}$.
	نشان دهید
	$A=B$.
	\\ \textbf{پاسخ:}
	عنصر دلخواه
	$x \in A$
	را در نظر بگیرید. داریم:
	$x = 5m+2 \ , \ m \in \mathbb{Z}$.
	اگر
	$m=k-1$،
	آنگاه
	$x = 5(k-1)+2 = 5k-3$.
	پس
	$A \subseteq B$.
	اکنون عنصر دلخواه
	$x \in B$
	را در نظر بگیرید. داریم:
	$x = 5m-3 \ , \ m \in \mathbb{Z}$.
	اگر
	$m=k+1$،
	آنگاه
	$x = 5(k+1)-3 = 5k+2$.
	پس
	$B \subseteq A$.
	بنابراین،
	$A=B$.
	\qed
\end{example}
\smallskip
مجموعه‌ی همه‌ی زیرمجموعه‌های $A$، مجموعه‌ی توانی $A$ نامیده می‌شود و با
$P(A)$
نشان داده می‌شود.
\begin{example}
	مجموعه‌ی توانی
	$A=\{x,y,z\}$
	را بنویسید.
	\\ \textbf{پاسخ:}
	$$P(A) = \{ \varnothing, \{x\}, \{y\}, \{z\}, \{x,y\}, \{x,z\}, \{y,z\}, \{x,y,z\} \}$$
\end{example}
\begin{example}
	مجموعه‌ی
	$P(P(P(\varnothing)))$
	را مشخص کنید.
	\\ \textbf{پاسخ:}
	$$P(\varnothing) = \{\varnothing\}$$
	$$P(P(\varnothing)) = \{\varnothing, \{\varnothing\}\}$$
	$$P(P(P(\varnothing))) = \{\varnothing, \{\varnothing\}, \{\{\varnothing\}\}, \{\varnothing, \{\varnothing\}\}\}$$
\end{example}
\begin{theorem}
	اگر 
	$n(A)=m$،
	آنگاه
	$n(P(A))=2^m$.
	\\ \dfth \\ \textbf{برهان:}
	\textcolor{magenta}{روش اول:}
	با استقرای ریاضی ثابت می‌کنیم.	اگر
	$m=0$،
	یعنی
	$A=\varnothing$،
	آنگاه
	$P(A)=\{\varnothing\}$،
	$2^0=1$
	عضو دارد. اکنون فرض می‌کنیم اگر مجموعه‌ی $A$ دارای
	$m=k$
	عضو باشد، آنگاه
	$P(A)$
	دارای
	$2^k$
	است. اکنون درستی ادعا را برای
	$m=k+1$
	نشان می‌دهیم. اگر به مجموعه‌ی $A$ یک عضو اضافه کنیم، مجموعه‌ی $A$،
	$k+1$
	عضوی می‌شود. این عضو جدید با
	$2^k$
	عضو دیگر مجموعه‌ی توانی،
	$2^k$
	عضو جدید برای مجموعه‌ی توانی پدید می‌آورد. در نتیجه، تعداد عضوهای مجموعه‌ی توانی مجموعه‌ی
	$k+1$
	عضوی، برابر با
	$2^k+2^k = 2^{k+1}$
	است. بنابراین، برهان کامل است.
	\qed \\
	\textcolor{magenta}{روش دوم:}
	برای هر زیرمجموعه‌ی $A$، دو احتمال برای هر عنصر
	$x \in A$
	وجود دارد، $x$ در زیرمجموعه باشد یا نباشد. در نتیجه، برای $m$ عنصر،
	$2^m$
	انتخاب مختلف برای ساختن یک زیرمجموعه از $A$ وجود خواهد داشت. بنابراین، تعداد زیرمجموعه‌های $A$ برابر با
	$2^m$
	است.
	\qed
\end{theorem}
\begin{example}
	تعداد زیرمجموعه‌های یک مجموعه‌ی
	$2n-5$
	عضوی، شانزده برابر تعداد زیرمجموعه‌های یک مجموعه‌ی
	$n-2$
	عضوی است. مقدار $n$ چقدر است؟
	\\ \textbf{پاسخ:}
	$$2^{2n-5} = 16 \times 2^{n-2} = 2^4 \times 2^{n-2} \Rightarrow 2^{2n-5} = 2^{n+2} \Rightarrow 2n-5=n+2 \Rightarrow n=7$$
\end{example}
\begin{example}
	سه مجموعه‌ی
	$n+3$
	عضوی،
	$n+1$
	عضوی و
	$n-2$
	عضوی روی هم ۱۶۴ زیرمجموعه دارند. مقدار $n$ چقدر است؟
	\\ \textbf{پاسخ:}
	$$2^{n+3} + 2^{n+1} + 2^{n-2} = 2^n(2^3+2^1+2^{-2}) = 164 \Rightarrow 2^n \times \frac{41}{4} = 64 \Rightarrow 2^n = 16 = 2^4 \Rightarrow n=4$$
\end{example}
\smallskip
\textbf{نمایش هندسی مجموعه‌ها:}
یکی از ابزارهای شهودی برای فهم بهتر مجموعه‌ها استفاده از تمثیل هندسی است. این کار برای اولین بار توسط ریاضی‌دانی به نام ون%
\LTRfootnote{ Venn}
انجام شد و نمودار ون نام گرفت. در نمودار ون در حالت کلی، عضوهای مجموعه درون یک خم بسته در صفحه مانند دایره یا مستطیل نشان داده می‌شود. گاهی ناحیه‌ی داخلی شکلی را که برای نمایش مجموعه به کار می‌رود هاشور می‌زنند. باید توجه داشت که نمودار ون نوعی تمثیل هندسی و شهودی است و به‌وسیله‌ی آن نمی‌توان یک قضیه‌ی ریاضی را ثابت کرد.
\\
\textbf{مجموعه‌ی مرجع:}
در یک بحث معین، تمام عضوهای مجموعه‌های مورد مطالعه متعلق به مجموعه‌ای موسوم به مجموعه‌ی مرجع است. به بیان دیگر، هر بحث دلخواه گفت‌وگویی است درباره‌ی زیرمجموعه‌های یک مجموعه‌ی مرجع. مجموعه‌ی مرجع را با حرف
$\mathbb{U}$
نشان می‌دهیم.

\section{جبر مجموعه‌ها}
\textbf{اجتماع:}
اجتماع مجموعه‌های $A$ و $B$ مجموعه‌ای است از همه‌ی عنصرهایی که متعلق به $A$ یا متعلق به $B$ (یا متعلق به هر دو مجموعه‌ی $A$ و $B$) باشند. اجتماع $A$ و $B$ را به صورت
$A \cup B$
نشان می‌دهیم.
$$A \cup B = \{x \ | \ x \in A \ \vee x \in B\}$$

اجتماع $n$ مجموعه‌ی
$A_1, A_2, \cdots, A_n$
را با نماد زیر نشان می‌دهیم.
$$\bigcup_{i=1}^{n}A_i$$
\begin{example}
	اگر
	$A=\{a,b,c,d,e\}$ و $B=\{a,e,i,o,u\}$،
	آنگاه
	$A \cup B = \{a,b,c,d,e,i,o,u\}$.
\end{example}
\begin{example}
	ثابت کنید
	$A \cup B = A$
	اگر و تنها اگر
	$B \subseteq A$.
	\\ \textbf{پاسخ:}
	اثبات
	($\Rightarrow$):
	فرض کنید
	$A \cup B = A$.
	اگر
	$B = \varnothing$
	آنگاه می‌دانیم
	$B \subseteq A$.
	اگر
	$B \neq \varnothing$
	آنگاه فرض کنید
	$x \in B$،
	در نتیجه
	$x \in A \cup B$.
	چون
	$A \cup B = A$،
	در نتیجه
	$x \in A$.
	پس اگر
	$x \in B$
	آنگاه
	$x \in A$.
	یعنی
	$B \subseteq A$. \\
	اثبات
	($\Leftarrow$):
	فرض کنید
	$B \subseteq A$.
	اگر
	$B = \varnothing$
	آنگاه می‌دانیم
	$B \cup A = \varnothing \cup A = A$.
	اگر
	$B \neq \varnothing$
	آنگاه اگر
	$x \in B$
	حتماً
	$x \in A$.
	در نتیجه
	$\{x\} \cup A = A$.
	اما این برای همه‌ی
	$x \in B$
	درست است، پس
	$B \cup A = A$. \\
	بنابراین اثبات کامل است.
	\qed
\end{example}
\smallskip
\textbf{اشتراک:}
اشتراک مجموعه‌های $A$ و $B$ مجموعه‌ای است از همه‌ی عنصرهایی که متعلق به $A$ و متعلق به $B$ باشند. اشتراک $A$ و $B$ را به صورت
$A \cap B$
نشان می‌دهیم.
$$A \cap B = \{x \ | \ x \in A \ \wedge x \in B\}$$

$$A \setminus B = \{x \ | \ x \in A \ \wedge \ x \notin B\}$$

$$A \Delta B = \{A \setminus B\} \cup (B \setminus A)$$

\begin{theorem}

	\vspace{-0.3cm}
	\begin{multicols}{2}
		\begin{enumerate}
			\item 
			$A \cup B = B \cup A$
			\item 
			$A \cap B = B \cap A$
			\item 
			$A \cup (B \cup B) = (A \cup B) \cup C$
			\item

			\item

			\item

			\item

			\item 
			$A \cup \mathbb{U} = \mathbb{U}$
			\item

			\item

			\item

			\item

			\item

			\item

			\item

			\item
