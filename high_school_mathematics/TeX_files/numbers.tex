\section{اصول موضوعه‌ی اعداد حقیقی}
مجموعه‌ی اعداد حقیقی را با نماد
$\mathbb{R}$
نشان می‌دهیم. در اینجا اعداد حقیقی را تعریف نمی‌کنیم، بلکه اصول موضوعه‌ای را برای آن بیان می‌کنیم. در این قسمت اصل‌های مربوط به جبر و در قسمت بعدی اصل‌های مربوط به ترتیب را بیان می‌کنیم. یک اصل دیگر (اصل وجود کوچک‌ترین کران بالا) باقی می‌ماند که بیان نمی‌کنیم و برای یادگیری آن می‌توانید به کتاب‌های مرجع مراجعه کنید.

دو عمل $+$ و
$\times$
که به‌ترتیب جمع و ضرب نامیده می‌شود، روی
$\mathbb{R}$
تعریف شده است. اگر $a$ و $b$ دو عدد حقیقی باشند، عمل ضرب
$a \times b$
را به صورت
$a \cdot b$
یا مختصراً
$ab$
نیز نمایش می‌دهند. همچنین علامت $=$ همواره به معنی «برابر است با» یا «این همان است» می‌باشد. علامت
$\neq$
نیز به معنی «برابر نیست با» یا «مخالف است با» می‌باشد.
\begin{enumerate}
	\item
	بسته‌بودن مجموعه‌ی
	$\mathbb{R}$
	نسبت به عمل جمع:
	$$\forall a,b \in \mathbb{R} \quad a+b \in \mathbb{R}$$
	\item
	بسته‌بودن مجموعه‌ی
	$\mathbb{R}$
	نسبت به عمل ضرب:
	$$\forall a,b \in \mathbb{R} \quad a \times b \in \mathbb{R}$$
	\item 
	جابه‌جایی عمل جمع:
	$$\forall a,b \in \mathbb{R} \quad a+b = b+a$$
	\item 
	جابه‌جایی عمل ضرب:
	$$\forall a,b \in \mathbb{R} \quad a \times b = b \times a$$
	\item 
	شرکت‌پذیری عمل جمع:
	$$\forall a,b,c \in \mathbb{R} \quad (a+b)+c = a+(b+c)$$
	\item 
	شرکت‌پذیری عمل ضرب:
	$$\forall a,b,c \in \mathbb{R} \quad (a \times b) \times c = a \times (b \times c)$$
	\item 
	عضو خنثای عمل جمع: فقط یک عدد حقیقی وجود دارد که برای عمل جمع اعداد حقیقی، خنثی است. این عدد را صفر می‌نامیم و آن را با نماد $0$ نشان می‌دهیم.
	$$\forall a \in \mathbb{R} \quad a+0 = 0+a = a$$
	\item 
	عضو خنثای عمل ضرب: فقط یک عدد حقیقی مخالف با صفر وجود دارد که برای عمل ضرب اعداد حقیقی، خنثی است. این عدد را یک می‌نامیم و آن را با نماد $1$ نشان می‌دهیم.
	$$\forall a \in \mathbb{R} \quad a \times 1 = 1 \times a = a$$
	\item 
	یک (۱) از صفر (۰) متمایز است.
	\item 
	وجود عدد قرینه: برای هر عدد $a$ فقط یک عدد وجود دارد که مجموع آن با $a$ مساوی است با صفر. این عدد را قرینه‌ی عدد $a$ می‌نامیم و آن را با نماد $-a$ نشان می‌دهیم.
	$$\forall a \in \mathbb{R} \quad a+(-a) = (-a)+a = 0$$
	\item 
	وجود وارون: برای هر عدد حقیقی $a$ مخالف صفر فقط یک عدد حقیقی وجود دارد که حاصل‌ضرب آن در $a$ مساوی است با یک. این عدد را وارون عدد $a$ می‌نامیم و آن را با نماد
	$\frac{1}{a}$ یا $a^{-1}$
	نشان می‌دهیم.
	$$\forall a \in \mathbb{R}-\{0\} \quad a \times (\frac{1}{a}) = (\frac{1}{a}) \times a = 1$$
	\item 
	خاصیت پخشی (توزیع‌پذیری) عمل ضرب نسبت به عمل جمع:
	$$\forall a,b,c \in \mathbb{R} \quad a \times (b+c) = (a \times b) + (a \times c)$$
	$$\forall a,b,c \in \mathbb{R} \quad (b+c) \times a = (b \times a) + (c \times a)$$
\end{enumerate}
\vspace{1mm}
\noindent
\textbf{خاصیت تساوی:}
اگر
$a=b$ و $c=d$،
آنگاه
$a+c=b+d$ و $a \times c=b \times d$.
\\
\textbf{عمل تفریق:}
اگر $a$ و $b$ دو عدد حقیقی باشند، مجموع عدد $a$ و قرینه‌ی عدد $b$ یعنی $-b$ را تفریق (تفاضل یا اختلاف) $a$ و $b$ می‌نامند و آن را با
$a-b$
نمایش می‌دهند. تفریق خاصیت جابه‌جایی ندارد و ترتیب در آن مهم است.
\\
\textbf{عمل تقسیم:}
اگر $a$ و $b$ دو عدد حقیقی باشند و
$b \neq 0$،
حاصل‌ضرب عدد $a$ در وارون عدد $b$ یعنی
$\frac{1}{b}$
را تقسیم $a$ بر $b$ می‌نامند و آن را با
$\frac{a}{b}$ یا $a \div b$
نمایش می‌دهند. تقسیم خاصیت جابه‌جایی ندارد و ترتیب در آن مهم است.
\vspace{2pt}
\begin{theorem}
	قرینه‌ی قرینه‌ی هر عدد حقیقی مساوی است با خود آن عدد، یعنی:
	$-(-a)=a$.
	\\ \dfth \\ \textbf{برهان:}
	بنا به اصل ۱۰ می‌دانیم قرینه‌ی $a$ برابر است با $-a$، یعنی مجموع $a$ و $-a$ برابر صفر است. از طرفی قرینه‌ی $-a$ یعنی	$-(-a)$ یکتاست، پس فقط یک عدد وجود دارد که جمع آن با $-a$ برابر صفر است. از قبل می‌دانیم که مجموع $a$ و $-a$ برابر صفر است. پس قرینه‌ی $-a$ برابر با $a$ است.
	\qed
\end{theorem}

\begin{theorem}
	قرینه‌ی مجموع دو عدد حقیقی مساوی است با مجموع قرینه‌های آن‌ها، یعنی:
	$-(a+b)=(-a)+(-b)$.
	\\ \dfth \\ \textbf{برهان:}
	بنا به اصل ۱۰ داریم
	$(a+b)+(-(a+b))=0 \ $.
	همچنین داریم:
	\begin{align*}
		(a+b)+((-a)+(-b)) &= a+b+(-a)+(-b) \\
		&= a+(-a)+b+(-b) &\text{اصل ۳} \\
		&= (a+(-a))+(b+(-b)) &\text{اصل ۵} \\
		&= 0+0 &\text{اصل ۱۰} \\
		&= 0
	\end{align*}
	با توجه به این‌که عضو قرینه یکتاست، بنابراین:
	$-(a+b)=(-a)+(-b)$.
	\qed
\end{theorem}

\begin{theorem} \label{nuthat0=0}
	برای هر $a$ داریم
	$a \times 0 = 0$.
	\\ \dfth \\ \textbf{برهان:}
	بنا به اصل ۷ داریم
	$1=1+0 \ $.
	بنابراین:
	\begin{align*}
		a \cdot 1 = a(1+0) &\Rightarrow a = (a \cdot 1) + (a \cdot 0) \\
		&\Rightarrow (-a)+a=(-a)+(a+ (a \cdot 0)) \\
		&\Rightarrow 0 = ((-a)+a)+(a \cdot 0) \\
		&\Rightarrow 0 = 0 + (a \cdot 0) \\
		&\Rightarrow 0 = a \cdot 0
	\end{align*}
	\qed
\end{theorem}

\begin{theorem}
	حاصل‌ضرب دو عدد حقیقی برابر با صفر است اگر و تنها اگر دست‌کم یکی از آن دو صفر باشد.
	$$\forall a,b \in \mathbb{R} ,\quad (a=0) \vee (b=0) \iff a \times b=0$$
	\dfth \\ \textbf{برهان:}
	اثبات
	($\Rightarrow$):
	بدون کاسته شدن از کلیت، فرض کنید
	$a=0$.
	بنا به قضیه‌ی
	\ref{nuthat0=0}،
	$a \times b = 0 \times b = 0$. \\
	اثبات
	($\Leftarrow$):
	فرض کنید
	$ab=0$.
	کافی است نشان دهیم که اگر
	$a \neq 0$
	آنگاه
	$b=0$.
	اگر
	$a \neq 0$
	آنگاه $a$ وارونی مانند
	$a^{-1}$
	دارد. بنابراین:
	$$a^{-1}(ab) = a^{-1} \cdot 0 = 0$$
	اما از طرفی داریم:
	$$a^{-1}(ab) = (a^{-1}a) \cdot b = 1 \cdot b = b$$
	بنابراین
	$b=0$. \\
	بنابراین اثبات کامل است.
	\qed
\end{theorem}

\begin{theorem}
	صفر هیچ وارونی ندارد. به بیان دیگر، عمل تقسیم بر صفر معنا ندارد.
	\\ \dfth \\ \textbf{برهان:}
	باید نشان دهیم هیچ عددی مانند $x$ وجود ندارد که
	$0 \times x = 1$.
	می‌دانیم که برای هر $x$،
	$0 \times x = 0$.
	اگر برای $x$ داشته باشیم
	$0 \times x = 1$،
	نتیجه می‌گیریم
	$0=1$
	و این بنا به اصل ۹ ناممکن است.
	\qed
\end{theorem}

\begin{theorem}
	اگر
	$a+b=a+c$،
	آنگاه
	$b=c$.
	\\ \dfth \\ \textbf{برهان:}
	اگر
	$a+b=a+c$،
	در آن صورت:
	$$(-a)+(a+b)=(-a)+(a+c) \Rightarrow (-a+a)+b=(-a+a)+c \Rightarrow 0+b=0+c \Rightarrow b=c$$
	\qed
\end{theorem}

\begin{theorem}
	اگر
	$ab=ac$ و $a \neq 0$،
	آنگاه
	$b=c$.
	\\ \dfth \\ \textbf{برهان:}
	اگر
	$ab=ac$ و $a \neq 0$،
	آنگاه $a$ دارای وارونی مانند
	$a^{-1}$
	است. بنابراین:
	$$a^{-1}(ab)=a^{-1}(ac) \Rightarrow (a^{-1}a) \cdot b = (a^{-1}a) \cdot c \Rightarrow 1 \cdot b=1 \cdot c \Rightarrow b=c$$
	\qed
\end{theorem}

\begin{theorem}
	برای هر $a$ و $b$، داریم
	$(-a)b = -(ab)$.
	\\ \dfth \\ \textbf{برهان:}
	کافی است نشان دهیم
	$-(a)b$
	قرینه‌ی
	$ab$
	است، یعنی
	$(-a)b+ab=0$.
	داریم:
	$$(-a)b+ab = ((-a)+a)b = 0 \times b = 0$$
	\qed
\end{theorem}

\begin{theorem}
	برای هر $a$ و $b$، داریم
	$(-a)(-b) = ab$.
	\\ \dfth \\ \textbf{برهان:}
	$$(-a)(-b) = -(a(-b)) = -((-b)a) = -(-(ba)) = ba = ab$$
	\qed
\end{theorem}

\begin{theorem}
	برای هر
	$a \neq 0$ و $b \neq 0$،
	داریم
	$(ab)^{-1}=a^{-1}b^{-1}$.
	\\ \dfth \\ \textbf{برهان:}
	کافی است نشان دهیم
	$a^{-1}b^{-1}$
	وارون
	$ab$
	است، یعنی
	$(ab)(a^{-1}b^{-1})=1$.
	$$(ab)(a^{-1}b^{-1}) = a(b(a^{-1}b^{-1})) = a(b(b^{-1}a^{-1})) = a((bb^{-1})a^{-1}) = a(1 \times a^{-1}) = aa^{-1} = 1$$
	\qed
\end{theorem}

\section{ترتیب در اعداد حقیقی}
رابطه‌ی
$a<b$
را در نظر بگیرید. اصل‌ها و قانون‌های حاکم بر این رابطه و ویژگی‌های آن به شرح زیر است.
\begin{enumerate}
	\item 
	اصل تثلیث: هر زوج از اعداد حقیقی مانند $a$ و $b$ در یک و فقط یکی از شرایط زیر صدق می‌کند:
	\begin{multicols}{3}
		\begin{enumerate}
			\item $a<b$
			\item $a=b$
			\item $a>b$
		\end{enumerate}
	\end{multicols}
	\item 
	تعدی: اگر
	$a<b$ و $b<c$،
	آنگاه
	$a<c$.
	\item 
	اگر
	$a>0$ و $b>0$،
	آنگاه
	$ab>0$.
	\item 
	اگر
	$a<b$،
	آنگاه برای هر $c$، داریم
	$a+c<b+c$.
	\item 
	$a<b \iff b>a$
	\item 
	$a \leq b \iff a<b \ \vee \ a=b$
	\item 
	$a<b<c \iff a<b \ \wedge \ b<c$
\end{enumerate}

\begin{theorem}
	اگر
	$a<b$ و $c<d$،
	آنگاه
	$a+c<b+d$.
	\\ \dfth \\ \textbf{برهان:}

	بنا به اصل تعدی:
	$$a+c<b+c \ \wedge \ b+c<b+d \Rightarrow a+c<b+d$$
	\qed
\end{theorem}

\section{اعداد طبیعی، صحیح، گویا و گنگ}
با شروع از ۰ و جمع کردن با ۱، عددهای طبیعی را می‌سازیم. چند عدد طبیعی ابتدا به صورت زیر است:
$$0 \ ,$$
$$0+1=1 \ ,$$
$$1+1=2 \ ,$$
$$2+1=3 \ ,$$
و به همین ترتیب هر عدد طبیعی را با جمع ۱ و عدد قبلی می‌سازیم. به بیان دیگر، عدد ۰ متعلق به مجموعه‌ی اعداد طبیعی است و اگر عدد $n$ به این مجموعه متعلق باشد، آنگاه
$n+1$
نیز به این مجموعه متعلق است.%
\RTLfootnote{ البته در اینجا ظرایفی وجود دارد که بهتر است با مراجعه به منابع معتبر مبانی ریاضیات آن را مطالعه کرد.}

مجموعه‌ی اعداد طبیعی%
\RTLfootnote{ در این جزوه، صفر را جزو اعداد طبیعی می‌دانیم. درباره‌ی سایر منابع به قرارداد خود آن منابع مراجعه کنید.}:
$$\mathbb{N} = \{0,1,2,3, \cdots\}$$
\begin{example}
	آیا تفریق اعداد طبیعی دارای خاصیت شرکت‌پذیری‌ هست یا خیر؟ تقسیم چطور؟
	\\ \textbf{برهان:}
	تفریق و تقسیم در مجموعه‌ی اعداد طبیعی خاصیت شرکت‌پذیری ندارند. مثلاً:
	$$(8-2)-3 \neq 8-(2-3)$$
	$$(12 \div 4) \div 3 \neq 12 \div (4 \div 3)$$
\end{example}
\smallskip
اگر تمام عددهای متعلق به
$\mathbb{N}$
و قرینه‌ی آن‌ها را در نظر بگیریم، مجموعه‌ی اعداد صحیح را ساخته‌ایم.
مجموعه‌ی اعداد صحیح:
$$\mathbb{Z} = \{\cdots, -3,-2,-1,0,1,2,3, \cdots\}$$
مجموعه‌ی اعداد صحیح مثبت:
$$\mathbb{Z}^+ = \{1,2,3,4,5 \cdots\}$$
مجموعه‌ی اعداد صحیح منفی:
$$\mathbb{Z}^- = \{\cdots, -5,-4,-3,-2,-1\}$$
مجموعه‌ی اعداد گویا:
$$\mathbb{Q} = \left\{\frac{m}{n} \ | \ m,n \in \mathbb{Z} \ \wedge \ n \neq 0\right\}$$
مجموعه‌ی اعداد گویای مثبت:
$$\mathbb{Q} = \left\{\frac{m}{n} \ | \ m,n \in \mathbb{Z}^+\right\}$$
مجموعه‌ی اعداد گویای منفی:
$$\mathbb{Q} = \left\{\frac{m}{n} \ | \ m,n \in \mathbb{Z} \ \wedge \ mn<0\right\}$$
